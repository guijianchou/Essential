    private static async Task VerifyFindingsVisibility(FrameworkElement root, DashboardPage audit, DataStorageService storage)
    {
        DateTime today = DateTime.Today;
        var result = SyntheticAudit(today.AddHours(2), true, 0);
        for (int index = 0; index < 9; index++)
        {
            bool update = index == 6, volsnap = index == 7;
            string log = index < 6 ? "Application" : "System";
            string provider = update ? "Microsoft-Windows-WindowsUpdateClient" : volsnap ? "Volsnap" : "SyntheticProvider";
            string eventId = update ? "20" : volsnap ? "36" : "120";
            string eventRef = log + ":" + (90000 + index);
            DateTime time = (update ? today.AddHours(1).AddMinutes(38).AddSeconds(47)
                : volsnap ? today.AddDays(-1).AddHours(2).AddMinutes(41).AddSeconds(13) : today.AddHours(1)).ToUniversalTime();
            var issue = SyntheticAudit(today, true, 1).Findings[0];
            issue.Key = "visibility_" + index;
            issue.Severity = index < 6 ? "High" : "Medium";
            issue.Category = log;
            issue.LogName = log;
            issue.Source = provider;
            issue.EventId = eventId;
            issue.EventRef = eventRef;
            issue.EventRecordId = (90000 + index).ToString();
            issue.EventTimestamp = time.ToString("O");
            issue.EventTimes = new() { [eventRef] = time };
            issue.RelatedEventRefs = new() { eventRef };
            issue.FirstSeenUtc = issue.LastSeenUtc = time;
            issue.SupportingEventCount = 1;
            issue.Title = update ? "Synthetic: Windows update failed" : volsnap ? "Synthetic: shadow copy failed" : "Synthetic priority finding " + index;
            issue.TitleZh = update ? "合成测试：Windows 更新失败" : volsnap ? "合成测试：卷影复制失败" : "合成测试：优先问题 " + index;
            result.Findings.Add(issue);
        }
        result.Metadata!["ScanStart"] = JsonSerializer.SerializeToElement(today.AddDays(-1).ToUniversalTime());
        result.Metadata["EventCount"] = result.Metadata["AnalyzedEventCount"] = JsonSerializer.SerializeToElement(9);
        result.Metadata["FilteredEventCount"] = JsonSerializer.SerializeToElement(0);
        result.Metadata["Channels"] = JsonSerializer.SerializeToElement(new[] { "Security", "System", "Application", "Setup", "ForwardedEvents" }
            .Select(log => new AuditChannelCoverage(log, log == "Security" ? "skipped" : "complete",
                log == "Application" ? 6 : log == "System" ? 3 : 0, log == "Security" ? "not_requested" : "none")));
        result.HealthScore = HealthScoreCalculator.Calculate(result.Findings).Score;
        await storage.SaveAuditResultAsync(result);
        await audit.ViewModel.LoadDataCommand.ExecuteAsync(null);
        var view = audit.ViewModel;
        view.SetFindingsViewCommand.Execute("Priority");
        await Task.Delay(150);
        var allList = (ItemsControl)audit.FindName("AllFindingsList");
        var priorityPanel = (FrameworkElement)audit.FindName("PriorityFindingsPanel");
        var allButton = (ToggleButton)audit.FindName("AllFindingsButton");
        var search = (TextBox)audit.FindName("FindingSearchBox");
        var scroll = (ScrollViewer)audit.FindName("AuditScroll");
        int total = view.FindingSections.Sum(section => section.Count);
        Require(total > 5 && view.PriorityFindings.Count == 5
            && view.PriorityFindings.All(issue => issue.Key is not ("visibility_6" or "visibility_7"))
            && view.PriorityPreviewLabel == AppText.Format("Showing {0} of {1} findings", 5, total)
            && priorityPanel.Visibility == Visibility.Visible && allList.Visibility == Visibility.Collapsed,
            "The priority preview explicitly counts five of all findings; both System errors remain outside it.");
        scroll.ChangeView(null, search.TransformToVisual(scroll).TransformPoint(default).Y + scroll.VerticalOffset - 12, null, true);
        await Task.Delay(100);
        await Snapshot(root, "findings-priority");
        ((IToggleProvider)new ToggleButtonAutomationPeer(allButton).GetPattern(PatternInterface.Toggle)).Toggle();
        await Task.Delay(200);
        Require(view.ShowAllFindings && allButton.IsChecked == true && allList.Visibility == Visibility.Visible
            && priorityPanel.Visibility == Visibility.Collapsed && view.FindingSections.All(section => section.IsExpanded),
            "The All findings button reveals every category without duplicating the priority preview.");
        foreach (string key in new[] { "visibility_6", "visibility_7" })
            Require(Descendants(allList).OfType<Button>().Any(button => button.Tag is AuditIssueEnhanced issue
                && issue.Key == key && button.ActualHeight > 0), "The complete list renders " + key + " beyond the first five.");
        view.FindingSections[0].IsExpanded = false;
        view.SetSourceFilterCommand.Execute("System");
        Require(view.FindingSections.All(section => section.IsExpanded)
            && view.FindingSections.SelectMany(section => section.Issues).All(issue => issue.LogName == "System"),
            "Selecting System opens matching categories even when previously collapsed.");
        view.FindingSections[0].IsExpanded = false;
        view.SetSourceFilterCommand.Execute("System");
        Require(view.FindingSections.All(section => section.IsExpanded), "Selecting the current source reopens its results.");
        view.FindingSections[0].IsExpanded = false;
        view.SetSeverityFilterCommand.Execute("Medium");
        Require(view.FindingSections.All(section => section.IsExpanded), "Severity filtering also opens matching categories.");
        string health = view.HealthScoreText;
        search.Text = "WindowsUpdateClient 20";
        await Task.Delay(150);
        Require(view.FindingSearchText == search.Text && view.FindingSections.SelectMany(section => section.Issues).Single().Key == "visibility_6"
            && view.ShowAllFindings && view.HealthScoreText == health,
            "Provider plus event-ID search finds the stored update failure without changing the audit score.");
        scroll.ChangeView(null, search.TransformToVisual(scroll).TransformPoint(default).Y + scroll.VerticalOffset - 12, null, true);
        await Task.Delay(100);
        await Snapshot(root, "findings-update");
        search.Text = "36";
        await Task.Delay(100);

    private static AuditResult SyntheticAudit(DateTime localTime, bool daily, int medium, bool partial = false)
    {
        var result = new AuditResult
        {
            Timestamp = localTime.ToUniversalTime(),
            Metadata = new Dictionary<string, JsonElement>
            {
                ["CoverageStatus"] = JsonSerializer.SerializeToElement(partial ? "partial" : daily ? "limited" : "complete"),
                ["AnalyzedEventCount"] = JsonSerializer.SerializeToElement(12),
                ["EventCount"] = JsonSerializer.SerializeToElement(12),
                ["AnalysisCompleted"] = JsonSerializer.SerializeToElement(true),
                ["AnalysisModel"] = JsonSerializer.SerializeToElement("gpt-5.6-luna"),
                ["ScanType"] = JsonSerializer.SerializeToElement("Full Scan"),
                ["ScanStart"] = JsonSerializer.SerializeToElement(localTime.AddHours(-1).ToUniversalTime()),
                ["ScanEnd"] = JsonSerializer.SerializeToElement(localTime.ToUniversalTime()),
                ["Channels"] = JsonSerializer.SerializeToElement(new[] { "Security", "System", "Application", "Setup", "ForwardedEvents" }
                    .Select(log => new AuditChannelCoverage(log, daily && log == "Security" ? "skipped" : "complete",
                        log == "System" ? 12 : 0, daily && log == "Security" ? "not_requested" : "none")))
            }
        };
        for (int i = 0; i < medium; i++)
            result.Findings.Add(new AuditIssue
            {
                Key = "synthetic_service_" + i, Severity = "Medium", Category = "System", LogName = "System",
                Title = "Synthetic service warning", TitleZh = "合成测试：服务告警",
                Description = "Synthetic evidence for UI regression.", DescriptionZh = "用于界面回归的合成证据。",
                RootCause = "The cause is not established.", RootCauseZh = "原因尚未确定。",
                Recommendation = "Review the service's next startup event.", RecommendationZh = "核对该服务下一次启动事件。",
                AnalysisModel = "gpt-5.6-luna"
            });
        result.HealthScore = HealthScoreCalculator.Calculate(result.Findings).Score;
        return result;
    }

    private static async Task Inspect(FrameworkElement root, string name)
    {
        root.UpdateLayout();
        var workflow = (FrameworkElement)root.FindName("WorkflowHost");
        var heading = (FrameworkElement)root.FindName("WorkflowHeading");
        var scroll = Descendants(workflow).OfType<ScrollViewer>().First();
        Log($"{name}: host={workflow.ActualWidth}x{workflow.ActualHeight}, viewport={scroll.ViewportWidth}x{scroll.ViewportHeight}, heading={heading.ActualWidth}x{heading.ActualHeight}, dc={workflow.DataContext?.GetType().Name}");
        Require(workflow.Visibility == Visibility.Visible && scroll.ViewportHeight > 40 && heading.ActualHeight > 0,
            name + ": workflow has a visible viewport.");
        Require(Microsoft.UI.Xaml.Hosting.ElementCompositionPreview.GetElementVisual(heading).Opacity > 0,
            name + ": heading is not transparent.");
        await Snapshot(root, name);
    }

    private static async Task Snapshot(FrameworkElement root, string name)
    {
        var bitmap = new RenderTargetBitmap();
        await bitmap.RenderAsync(root);
        using (var file = File.Create(Path.Combine(TestRoot, name + ".png")))
        {
            var encoder = await BitmapEncoder.CreateAsync(BitmapEncoder.PngEncoderId, file.AsRandomAccessStream());
            encoder.SetPixelData(BitmapPixelFormat.Bgra8, BitmapAlphaMode.Premultiplied,
                (uint)bitmap.PixelWidth, (uint)bitmap.PixelHeight, 96, 96, (await bitmap.GetPixelsAsync()).ToArray());
            await encoder.FlushAsync();
        }
    }

    private static System.Collections.Generic.IEnumerable<DependencyObject> Descendants(DependencyObject parent)
    {
        for (int i = 0; i < VisualTreeHelper.GetChildrenCount(parent); i++)
        {
            var child = VisualTreeHelper.GetChild(parent, i);
            yield return child;
            foreach (var descendant in Descendants(child)) yield return descendant;
        }
    }

    private static void Require(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
        Log("PASS " + message);
    }
    private static void Log(string message) => File.AppendAllText(Path.Combine(TestRoot, "result.txt"), message + Environment.NewLine);
